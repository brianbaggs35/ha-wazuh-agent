#include <tunables/global>

# AppArmor profile for the Wazuh Agent app.
#
# Shipping a profile is worth +1 on the Supervisor's security rating. It follows the structure of
# the Home Assistant default app profile on purpose.
#
# This app is the Wazuh agent of the Home Assistant OS device itself, not of its own container: its
# daemons run inside the host's file system (see wazuh-ha-host), read what the host has (processes,
# files, logs), execute the host's commands when a user configures one, and are started from a
# private mount namespace the app builds. So the profile is permissive about files, capabilities,
# execution and mounts (the container's capability set is the limit, and config.yaml adds only
# SYS_ADMIN and SYS_PTRACE), and restricts the parts that still make sense:
#
#   * signals go only to the app's own processes: with the host's process namespace shared, a PID is
#     a PID on the whole device, and this app has no business signalling any other
#   * ptrace may read other processes' /proc entries, which is how the host is looked at, but may not
#     attach to anything but the app's own processes
#   * no access to the kernel's memory image or the sysrq trigger
#
# The profile name must equal the app slug.
profile wazuh_agent flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities (limited by the container's own set; config.yaml adds SYS_ADMIN and SYS_PTRACE)
  capability,
  file,
  network,

  # Mounts: the agent runs in a private copy of the host's mount namespace, with a few of the host's
  # files covered by copies (wazuh-ha-host and hostns-start), which takes mounting.
  mount,
  umount,

  # Signals: the app and its daemons signal each other; Docker stops the container by signalling its
  # init process. Nothing else is signalled.
  signal (send,receive) peer=wazuh_agent,
  signal (receive) peer=unconfined,

  # ptrace: reading /proc/<pid>/{root,ns,exe,fd} of the host's processes (system inventory, the host's
  # file system) is "read". Attaching ("trace") is for the app's own processes only.
  ptrace (read,readby),
  ptrace (trace,tracedby) peer=wazuh_agent,

  # Everything that runs here inherits this profile: the app's scripts, the Wazuh daemons (from the
  # data volume, which the host sees as /mnt/data/...), and the host's own commands that the agent
  # runs on request (commands to monitor, active response).
  /** ix,

  # Nothing here has a use for these.
  deny /proc/sysrq-trigger rwklx,
  deny /proc/kcore rwklx,
}
